The core logic for implementing the smart search feature.

# Relevant Flask/Python Snippet for Search Logic

@app.route('/api/items', methods=['GET'])
def get_items():
    query = request.args.get('q', '').lower()
    items = load_data()
    
    if not query:
        return jsonify(items)
        
    filtered_items = [
        item for item in items 
        if query in item['title'].lower() or 
           query in item['creator'].lower()
    ]
    return jsonify(filtered_items)
